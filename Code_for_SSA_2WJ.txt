def gillespie_simulation(seq_k1, initial_counts,t_max: float):
    result_output = []
    results_list = []
    counts = initial_counts.copy()
    t = 0.0
    seq_reaction = 0
    next_save_time = 0.025
    reactions = generate_reactions(seq_k1, initial_counts)

    results_list.append(np.hstack([t, counts]))
    print(f"start: Time: {t:.1f}, A: {(counts[0]/6.02e23/1e-11):.1e}, B: {(counts[1]/6.02e23/1e-11):.1e}")


    while t < t_max:
        reaction_1 = generate_reactions(seq_k1, counts)
        sum_column2 = sum(row[1] for row in reaction_1)

        if sum_column2 <= 1e-12:
            print("所有反应倾向为零，模拟终止")
            print ('seq:',seq_reaction/1e3, 't:',t)
            results_list.append(np.hstack([t, counts]))
            result_output = np.array(results_list)
            break

        tau = -math.log(random.random()) / sum_column2
        t += tau
        seq_reaction += 1

        if t >= next_save_time:
            print ('seq:',seq_reaction/5e4, 't:',t)
            results_list.append(np.hstack([t, counts]))
            result_output = np.array(results_list)

            if t < 30:
                next_save_time += 0.025
            elif t < 200:
                next_save_time += 1
            elif t < 2000:
                next_save_time += 10

            else:
                next_save_time += 100


        r = random.random() * sum_column2
        cum_sum = 0.0
        for i in range(len(reactions)):
            cum_sum += reaction_1[i][1]
            if cum_sum >= r:
                delta = reactions[i][2]
                new_counts = [x + y for x, y in zip(counts, delta)]
                if all(c >= 0 for c in new_counts):
                    counts = new_counts
                break
    return result_output
